package com.mlomega.xr.securesurface;

import android.content.Context;
import android.hardware.display.DisplayManager;
import android.hardware.display.VirtualDisplay;
import android.os.Binder;
import android.os.IBinder;
import android.os.Parcel;
import android.os.Process;
import android.os.SystemClock;
import android.util.Log;
import android.view.InputDevice;
import android.view.InputEvent;
import android.view.MotionEvent;
import android.view.Surface;

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.lang.reflect.Method;

/**
 * Minimal Shizuku UserService used only by the isolated secure-surface spike.
 * It runs as uid=shell, creates a secure/trusted VirtualDisplay backed directly
 * by the XREAL protected Surface, and launches one public YouTube intent there.
 */
public final class TrustedDisplayUserService extends Binder {
    static final String DESCRIPTOR =
            "com.mlomega.xr.securesurface.ITrustedDisplayUserService";
    static final int TRANSACTION_CREATE = IBinder.FIRST_CALL_TRANSACTION;
    static final int TRANSACTION_LAUNCH_YOUTUBE = IBinder.FIRST_CALL_TRANSACTION + 1;
    static final int TRANSACTION_RELEASE = IBinder.FIRST_CALL_TRANSACTION + 2;
    static final int TRANSACTION_POINTER = IBinder.FIRST_CALL_TRANSACTION + 3;
    static final int TRANSACTION_HOVER = IBinder.FIRST_CALL_TRANSACTION + 4;
    static final int TRANSACTION_OPEN_KEYBOARD = IBinder.FIRST_CALL_TRANSACTION + 5;

    private static final String TAG = "MLOmegaTrustedVD";
    private Context suppliedContext;
    private VirtualDisplay virtualDisplay;
    private Surface displaySurface;
    private Object inputManager;
    private Method injectInputEvent;
    private Method setMotionDisplayId;
    private boolean inputReflectionBroken;

    @SuppressWarnings("unused")
    public TrustedDisplayUserService() {
        this(null);
    }

    @SuppressWarnings("unused")
    public TrustedDisplayUserService(Context context) {
        suppliedContext = context;
        attachInterface(null, DESCRIPTOR);
        setupInputInjection();
        Log.i(TAG, "service created uid=" + Process.myUid());
    }

    @Override
    protected boolean onTransact(int code, Parcel data, Parcel reply, int flags) {
        try {
            if (code == INTERFACE_TRANSACTION) {
                reply.writeString(DESCRIPTOR);
                return true;
            }
            data.enforceInterface(DESCRIPTOR);
            if (code == TRANSACTION_CREATE) {
                String name = data.readString();
                int width = data.readInt();
                int height = data.readInt();
                int dpi = data.readInt();
                int displayFlags = data.readInt();
                Surface surface = Surface.CREATOR.createFromParcel(data);
                int displayId = createDisplay(
                        name, width, height, dpi, displayFlags, surface);
                reply.writeNoException();
                reply.writeInt(displayId);
                return true;
            }
            if (code == TRANSACTION_LAUNCH_YOUTUBE) {
                int displayId = data.readInt();
                String uri = data.readString();
                String result = launchYoutube(displayId, uri);
                reply.writeNoException();
                reply.writeString(result);
                return true;
            }
            if (code == TRANSACTION_RELEASE) {
                releaseDisplay();
                reply.writeNoException();
                return true;
            }
            if (code == TRANSACTION_POINTER) {
                int displayId = data.readInt();
                int action = data.readInt();
                float x = data.readFloat();
                float y = data.readFloat();
                long downTime = data.readLong();
                boolean injected = injectPointer(
                        displayId, action, x, y, downTime);
                reply.writeNoException();
                reply.writeInt(injected ? 1 : 0);
                return true;
            }
            if (code == TRANSACTION_HOVER) {
                int displayId = data.readInt();
                float x = data.readFloat();
                float y = data.readFloat();
                boolean injected = injectHover(displayId, x, y);
                reply.writeNoException();
                reply.writeInt(injected ? 1 : 0);
                return true;
            }
            if (code == TRANSACTION_OPEN_KEYBOARD) {
                int displayId = data.readInt();
                boolean opened = openKeyboard(displayId);
                reply.writeNoException();
                reply.writeInt(opened ? 1 : 0);
                return true;
            }
            return super.onTransact(code, data, reply, flags);
        } catch (Throwable error) {
            Log.e(TAG, "transaction " + code + " failed", error);
            reply.writeException(new IllegalStateException(
                    error.getClass().getSimpleName() + ": " + error.getMessage()));
            return true;
        }
    }

    private int createDisplay(
            String name,
            int width,
            int height,
            int dpi,
            int flags,
            Surface surface) throws Exception {
        releaseDisplay();
        long identity = Binder.clearCallingIdentity();
        try {
            Context context = obtainShellContext();
            if (context == null) {
                throw new IllegalStateException("shell context unavailable");
            }
            DisplayManager manager = (DisplayManager) context.getSystemService(
                    Context.DISPLAY_SERVICE);
            if (manager == null) {
                throw new IllegalStateException("DisplayManager unavailable");
            }
            virtualDisplay = manager.createVirtualDisplay(
                    name,
                    width,
                    height,
                    dpi,
                    surface,
                    flags);
            if (virtualDisplay == null || virtualDisplay.getDisplay() == null) {
                throw new IllegalStateException("trusted VirtualDisplay creation failed");
            }
            displaySurface = surface;
            int id = virtualDisplay.getDisplay().getDisplayId();
            Log.i(TAG, "created display=" + id + " flags=0x"
                    + Integer.toHexString(flags) + " uid=" + Process.myUid());
            return id;
        } finally {
            Binder.restoreCallingIdentity(identity);
        }
    }

    private String launchYoutube(int displayId, String uri) throws Exception {
        long identity = Binder.clearCallingIdentity();
        try {
            ProcessBuilder builder = new ProcessBuilder(
                    "/system/bin/am",
                    "start",
                    "--display", String.valueOf(displayId),
                    "-f", "0x18000000",
                    "-a", "android.intent.action.VIEW",
                    "-d", uri,
                    "-p", "com.google.android.youtube");
            builder.redirectErrorStream(true);
            java.lang.Process process = builder.start();
            StringBuilder output = new StringBuilder();
            try (BufferedReader reader = new BufferedReader(
                    new InputStreamReader(process.getInputStream()))) {
                String line;
                while ((line = reader.readLine()) != null) {
                    if (output.length() > 0) output.append(' ');
                    output.append(line);
                }
            }
            int exit = process.waitFor();
            String result = "exit=" + exit + " " + output;
            Log.i(TAG, "YouTube launch display=" + displayId + " -> " + result);
            return result;
        } finally {
            Binder.restoreCallingIdentity(identity);
        }
    }

    private Context obtainShellContext() {
        try {
            Class<?> activityThread = Class.forName("android.app.ActivityThread");
            Object thread = activityThread.getMethod("currentActivityThread").invoke(null);
            if (thread != null) {
                Context system = (Context) activityThread
                        .getMethod("getSystemContext")
                        .invoke(thread);
                if (system != null) {
                    String[] packages = system.getPackageManager()
                            .getPackagesForUid(Process.myUid());
                    if (packages != null && packages.length > 0) {
                        return system.createPackageContext(
                                packages[0], Context.CONTEXT_IGNORE_SECURITY);
                    }
                    return system;
                }
            }
        } catch (Throwable error) {
            Log.w(TAG, "system context fallback", error);
        }
        return suppliedContext;
    }

    private void setupInputInjection() {
        try {
            try {
                Class<?> global = Class.forName(
                        "android.hardware.input.InputManagerGlobal");
                inputManager = global.getMethod("getInstance").invoke(null);
                injectInputEvent = global.getMethod(
                        "injectInputEvent", InputEvent.class, int.class);
            } catch (Throwable modernUnavailable) {
                Class<?> legacy = Class.forName(
                        "android.hardware.input.InputManager");
                inputManager = legacy.getMethod("getInstance").invoke(null);
                injectInputEvent = legacy.getMethod(
                        "injectInputEvent", InputEvent.class, int.class);
            }
            setMotionDisplayId = MotionEvent.class.getMethod(
                    "setDisplayId", int.class);
            inputReflectionBroken = inputManager == null ||
                    injectInputEvent == null || setMotionDisplayId == null;
            Log.i(TAG, "input injection ready=" + !inputReflectionBroken);
        } catch (Throwable error) {
            inputReflectionBroken = true;
            Log.e(TAG, "input injection reflection unavailable", error);
        }
    }

    private boolean injectPointer(
            int displayId,
            int action,
            float x,
            float y,
            long downTime) {
        if (inputReflectionBroken || displayId < 0) return false;
        long now = SystemClock.uptimeMillis();
        long gestureStart = downTime > 0L ? downTime : now;
        MotionEvent event = MotionEvent.obtain(
                gestureStart,
                now,
                action,
                x,
                y,
                0);
        try {
            event.setSource(InputDevice.SOURCE_TOUCHSCREEN);
            setMotionDisplayId.invoke(event, displayId);
            Object result = injectInputEvent.invoke(inputManager, event, 0);
            return !(result instanceof Boolean) || (Boolean) result;
        } catch (Throwable error) {
            Log.e(TAG, "pointer injection failed action=" + action +
                    " display=" + displayId + " x=" + x + " y=" + y,
                    error);
            return false;
        } finally {
            event.recycle();
        }
    }

    private boolean injectHover(int displayId, float x, float y) {
        if (inputReflectionBroken || displayId < 0) return false;
        long now = SystemClock.uptimeMillis();
        MotionEvent.PointerProperties properties =
                new MotionEvent.PointerProperties();
        properties.id = 0;
        properties.toolType = MotionEvent.TOOL_TYPE_MOUSE;
        MotionEvent.PointerCoords coordinates = new MotionEvent.PointerCoords();
        coordinates.x = x;
        coordinates.y = y;
        coordinates.pressure = 0f;
        coordinates.size = 1f;
        MotionEvent event = MotionEvent.obtain(
                now,
                now,
                MotionEvent.ACTION_HOVER_MOVE,
                1,
                new MotionEvent.PointerProperties[] { properties },
                new MotionEvent.PointerCoords[] { coordinates },
                0,
                0,
                1f,
                1f,
                0,
                0,
                InputDevice.SOURCE_MOUSE,
                0);
        try {
            setMotionDisplayId.invoke(event, displayId);
            Object result = injectInputEvent.invoke(inputManager, event, 0);
            return !(result instanceof Boolean) || (Boolean) result;
        } catch (Throwable error) {
            Log.e(TAG, "pointer hover injection failed display=" + displayId,
                    error);
            return false;
        } finally {
            event.recycle();
        }
    }

    private boolean openKeyboard(int displayId) {
        if (displayId < 0) return false;
        try {
            ProcessBuilder builder = new ProcessBuilder(
                    "/system/bin/input",
                    "-d", String.valueOf(displayId),
                    "keyevent", "KEYCODE_SEARCH");
            builder.redirectErrorStream(true);
            java.lang.Process process = builder.start();
            int exit = process.waitFor();
            Log.i(TAG, "keyboard/search display=" + displayId + " exit=" + exit);
            return exit == 0;
        } catch (Throwable error) {
            Log.e(TAG, "keyboard/search failed display=" + displayId, error);
            return false;
        }
    }

    private void releaseDisplay() {
        if (virtualDisplay != null) {
            try { virtualDisplay.setSurface(null); } catch (Throwable ignored) {}
            try { virtualDisplay.release(); } catch (Throwable ignored) {}
            virtualDisplay = null;
        }
        if (displaySurface != null) {
            try { displaySurface.release(); } catch (Throwable ignored) {}
            displaySurface = null;
        }
    }

    @SuppressWarnings("unused")
    public void destroy() {
        releaseDisplay();
        System.exit(0);
    }
}
